    def arm_callback(self, request, response):
        request # request must be specified even if it is not used
        self.get_logger().info('"%s" request to arm!' %request.data)
        try:
            self.get_logger().info("Sending arm request...")
            # Call the service asynchronously
            self.get_logger().info(f"Sending arm {request.data} request...")
            self.arm_req.value = request.data
           
            future = self.arming.call_async(self.arm_req)
            # Add a done callback to handle the result asynchronously
            future.add_done_callback(lambda f: self.handle_arm_future(f, response))
            response.success = True
            response.message = f"Arming: request recieved {request.data}"
            self.stop = request.data
            
        except Exception as e:
            self.get_logger().error(f"Service call failed: {e}")
            response.success = False
            response.message = "Arming: error during service call"
            return response
        
        return response
    

    def set_mode_callback(self, request, response):
        """
        Sets the drone mode to "GUIDED" and fills the response asynchronously after the service call completes.
        """
        request
        # Create the mode request
        mode_request = SetMode.Request()
        mode_request.custom_mode = 'GUIDED'  
        try:
            self.get_logger().info("Sending mode request...")
            # Call the service asynchronously
            future = self.set_mode_client.call_async(mode_request)
            # Add a done callback to handle the result asynchronously
            future.add_done_callback(lambda f: self.handle_set_mode_future(f, response))
            response.success = True
            response.message = "Switch Mode: request recieved"
            
        except Exception as e:
            self.get_logger().error(f"Service call failed: {e}")
            response.success = False
            response.message = "Switch Mode: error during service call"
            return response
        
        self.get_logger().info("Sent mode request.")
        return response   
         


    def handle_set_mode_future(self, future, response):
        """
        Processes the result of the `set_mode_client` future and sets the response.
        """
        try:
            result = future.result()  # Get the result of the service call
            if result is not None and result.mode_sent:  # Check the response structure
                response.success = True
                response.message = "Switch Mode: success"
                self.get_logger().info("Success")
            else:
                response.success = False
                response.message = "Switch Mode: failed"
                self.get_logger().warn("Fail")

        except Exception as e:
            self.get_logger().error(f"Error processing future: {e}")
            response.success = False
            response.message = "Switch Mode: error during response handling"

    def handle_arm_future(self, future, response):
        try:
            result = future.result()  # Get the result of the service call
            if result is not None and result.success:  # Check the response structure
                response.success = True
                response.message = "Arming: success"
                self.get_logger().info("Success")
            else:
                response.success = False
                response.message = "Arming: failed"
                self.get_logger().warn("Fail")

        except Exception as e:
            self.get_logger().error(f"Error processing future: {e}")
            response.success = False
            response.message = "Arming: error during response handling"



    def arm(self):
        # Arms the drone
        arm_request = CommandBool.Request()
        arm_request.value = True 
        try:
            self.get_logger().warn("sending arm request")
            future = self.arming.call_async(arm_request)
            rclpy.spin_until_future_complete(self, future, timeout_sec=1)  # Wait for the service response
            if future.result() is not None and future.result().success:
                self.get_logger().warn("Drone armed")
                self.current_req = True
            else:
                self.get_logger().error("Failed to arm the drone or no response")
        except Exception as e:
            self.get_logger().error(f"Service call failed: {e}")

    def takeoff(self):
        """
        The takeoff strategy for the drone.
        """
        # Arm the drone
        if not self.armed:
            self.arm()

        # Takeoff
        request = self.takeoff_req
        request.altitude = 0.5
        request.latitude = 0.
        request.longitude = 0.
        request.min_pitch = 0.
        request.yaw = 0.
        
        try:
            self.get_logger().warn("sending takeoff request")
            future = self.takeoff_client.call_async(request)
            rclpy.spin_until_future_complete(self, future,timeout_sec=10)  # Wait for the service response
            if future.result() is not None:
                resp = future.result()
                if resp.success:
                    self.get_logger().warn("takeoff successful")
                    self.current_req = True
            else:
                self.get_logger().error("Service call failed")

        except Exception as e:
            self.get_logger().error(f"Service call failed: {e}")

        self.get_logger().warn("takeoff END")
